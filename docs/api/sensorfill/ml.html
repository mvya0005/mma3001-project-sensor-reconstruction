sensorfill.ml

Machine-learning regression imputers (unit topics 5.1-5.6).

Each model learns the residual between the truth and linear interpolation from gap-aware features (sensorfill.features). Models:

  • linear - ordinary least squares on standardised features (5.1)
  • ridge - L2-regularised least squares (overfitting control, 3.2/5.6)
  • tree - CART decision tree (5.2)
  • forest - random forest (bagged trees; variance reduction)
  • svr - epsilon-SVR with an RBF kernel (5.3)
  • mlp - multilayer perceptron with early stopping (5.4)

Hyperparameters are selected by grouped K-fold cross-validation in which all rows from one gap stay in the same fold (rows inside a gap are strongly correlated; splitting them would leak information and overstate skill).

Example:
>>> from sensorfill.ml import make_model
>>> list(make_model("ridge").named_steps)
['scale', 'reg']

 42def make_model(name: str, random_state: int = 0, **params) -> Pipeline:
 43    """Create a scikit-learn pipeline (scaler + regressor) by name.
 44
 45    Args:
 46        name: One of ``linear, ridge, tree, forest, svr, mlp``.
 47        random_state: Seed for stochastic models.
 48        **params: Overrides passed to the regressor.
 49
 50    Returns:
 51        Unfitted :class:`~sklearn.pipeline.Pipeline` whose last step is named ``"reg"``.
 52
 53    Raises:
 54        ValueError: If ``name`` is not a known model.
 55
 56    Example:
 57        >>> make_model("tree", max_depth=3).named_steps["reg"].max_depth
 58        3
 59    """
 60    if name == "linear":
 61        reg = LinearRegression(**params)
 62    elif name == "ridge":
 63        reg = Ridge(**({"alpha": 1.0} | params))
 64    elif name == "tree":
 65        reg = DecisionTreeRegressor(**({"max_depth": 8, "min_samples_leaf": 20, "random_state": random_state} | params))
 66    elif name == "forest":
 67        reg = RandomForestRegressor(
 68            **({"n_estimators": 200, "min_samples_leaf": 5, "n_jobs": -1, "random_state": random_state} | params)
 69        )
 70    elif name == "svr":
 71        reg = SVR(**({"C": 10.0, "epsilon": 1.0, "gamma": "scale"} | params))
 72    elif name == "mlp":
 73        reg = MLPRegressor(
 74            **(
 75                {
 76                    "hidden_layer_sizes": (64, 64),
 77                    "alpha": 1e-3,
 78                    "learning_rate_init": 1e-3,
 79                    "early_stopping": True,
 80                    "max_iter": 500,
 81                    "random_state": random_state,
 82                }
 83                | params
 84            )
 85        )
 86    else:
 87        raise ValueError(f"unknown model {name!r}")
 88    # Trees do not need scaling but it is harmless; one pipeline keeps the API uniform.
 89    return Pipeline([("scale", StandardScaler()), ("reg", reg)])
 90
 91
 92#: Default hyperparameter grids searched by :func:`tune`.
 93PARAM_GRIDS = {
 94    "linear": {},
 95    "ridge": {"reg__alpha": [0.1, 1, 10, 100]},
 96    "tree": {"reg__max_depth": [4, 6, 8, 12, None], "reg__min_samples_leaf": [5, 20, 50]},
 97    "forest": {"reg__max_depth": [8, 16, None], "reg__min_samples_leaf": [2, 5, 20]},
 98    "svr": {"reg__C": [1, 10, 100], "reg__epsilon": [0.5, 2.0], "reg__gamma": ["scale", 0.05]},
 99    "mlp": {"reg__hidden_layer_sizes": [(32,), (64, 64), (128, 64, 32)], "reg__alpha": [1e-4, 1e-2]},
100}

103def subsample(X, y, groups, max_rows: int | None, rng=0):
104    """Randomly keep whole gaps until at least ``max_rows`` rows are kept.
105
106    Needed because SVR training costs O(n^2)-O(n^3) in the number of rows.
107    Whole gaps are kept (never split) so grouped cross-validation stays valid.
108
109    Args:
110        X: Feature table.
111        y: Targets, one per row of ``X``.
112        groups: Gap id of each row.
113        max_rows: Row budget; ``None`` keeps everything.
114        rng: Seed or generator.
115
116    Returns:
117        ``(X, y, groups)`` restricted to the kept gaps.
118
119    Example:
120        >>> X = pd.DataFrame({"a": range(6)})
121        >>> y = np.arange(6.0)
122        >>> g = np.array([0, 0, 1, 1, 2, 2])
123        >>> len(subsample(X, y, g, max_rows=3)[1])
124        4
125    """
126    if max_rows is None or len(y) <= max_rows:
127        return X, y, groups
128    rng = np.random.default_rng(rng)
129    ug = rng.permutation(np.unique(groups))
130    sizes = pd.Series(groups).value_counts()
131    keep, total = [], 0
132    for g in ug:
133        if total >= max_rows:
134            break
135        keep.append(g)
136        total += sizes[g]
137    m = np.isin(groups, keep)
138    return X[m], y[m], groups[m]

141def tune(name: str, X, y, groups, n_splits: int = 5, max_rows: int | None = None, grid: dict | None = None, rng=0):
142    """Grouped K-fold grid search for one model.
143
144    Args:
145        name: Model name (see :func:`make_model`).
146        X: Feature table from :func:`sensorfill.features.make_training_set`.
147        y: Residual targets from the same call.
148        groups: Gap ids from the same call (keep a gap's rows in one fold).
149        n_splits: Number of folds.
150        max_rows: Optional row cap (whole gaps kept) for expensive models.
151        grid: Override of :data:`PARAM_GRIDS`.
152        rng: Seed for subsampling.
153
154    Returns:
155        Fitted :class:`~sklearn.model_selection.GridSearchCV` (``best_estimator_``
156        is refit on the full, possibly subsampled, training set).
157
158    Raises:
159        KeyError: If ``name`` has no default grid and ``grid`` is None.
160
161    Example:
162        >>> rng = np.random.default_rng(0)
163        >>> X = pd.DataFrame({"a": rng.normal(size=60)})
164        >>> y = 2 * X["a"].to_numpy()
165        >>> gs = tune("ridge", X, y, np.repeat(np.arange(12), 5), n_splits=3)
166        >>> gs.best_params_
167        {'reg__alpha': 0.1}
168    """
169    X, y, groups = subsample(X, y, groups, max_rows, rng)
170    grid = PARAM_GRIDS[name] if grid is None else grid
171    gs = GridSearchCV(
172        make_model(name),
173        grid or {"reg__fit_intercept": [True]},
174        cv=GroupKFold(n_splits=n_splits),
175        scoring="neg_root_mean_squared_error",
176        n_jobs=-1,
177        refit=True,
178        return_train_score=True,
179    )
180    gs.fit(X, y, groups=groups)
181    return gs

184@dataclass
185class MLImputer:
186    """Robust linear baseline plus a learned residual correction.
187
188    Attributes:
189        model: Fitted regressor (pipeline) predicting ``y - base``.
190        builder: Feature builder for the grid being imputed.
191        clip: Optional ``(low, high)`` physical limits applied to filled values.
192        fit_seconds: Training time recorded by :meth:`fit`.
193
194    Example:
195        >>> ts = pd.date_range("2024-06-03", periods=6, freq="10min")
196        >>> b = GapFeatureBuilder(ts, use_neighbour_days=False, anchor_k=1)
197        >>> X, _, _ = b.build([0.0, np.nan, 2.0, np.nan, 4.0, 5.0])
198        >>> imp = MLImputer(make_model("linear"), b).fit(X, np.zeros(len(X)))
199        >>> imp.impute([0.0, np.nan, 2.0, np.nan, 4.0, 5.0])  # zero correction = linear
200        array([0., 1., 2., 3., 4., 5.])
201    """
202
203    model: object
204    builder: GapFeatureBuilder
205    clip: tuple | None = None
206    fit_seconds: float = field(default=0.0)
207
208    def fit(self, X, target):
209        """Fit the residual model and record the wall-clock training time.
210
211        Args:
212            X: Feature table (from :func:`sensorfill.features.make_training_set`).
213            target: Residuals ``y_true - base`` for each row.
214
215        Returns:
216            ``self``, so calls can be chained.
217        """
218        t0 = time.perf_counter()
219        self.model.fit(X, target)
220        self.fit_seconds = time.perf_counter() - t0
221        return self
222
223    def impute(self, y) -> np.ndarray:
224        """Fill the gaps of a series.
225
226        Args:
227            y: Series with NaNs on the builder's grid.
228
229        Returns:
230            Filled copy of ``y``: baseline plus learned correction inside
231            interior gaps; edge gaps fall back to the baseline. Filled values
232            are clipped to :attr:`clip` if set; observed values are untouched.
233
234        Raises:
235            ValueError: If ``y`` does not match the builder's grid length.
236        """
237        X, idx, y_lin = self.builder.build(y)
238        out = y_lin.copy()
239        if len(idx):
240            out[idx] = y_lin[idx] + self.model.predict(X)
241        if self.clip is not None:
242            lo, hi = self.clip
243            miss = np.isnan(np.asarray(y, dtype=float))
244            out[miss] = np.clip(out[miss], lo, hi)
245        return out

Fill the gaps of a series.

Arguments:
  • y: Series with NaNs on the builder's grid.
Returns:

Filled copy of y: baseline plus learned correction inside interior gaps; edge gaps fall back to the baseline. Filled values are clipped to clip if set; observed values are untouched.

Raises:
  • ValueError: If y does not match the builder's grid length.
