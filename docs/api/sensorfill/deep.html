sensorfill.deep

Deep-learning imputers in PyTorch (unit topics 5.4, 10.2, 10.3).

Designed to train on a Colab GPU (device="cuda") but runs on CPU too.

  • BiGRUImputer - bidirectional GRU that reads a window around each gap (observed values, the mask, occupancy and calendar channels) and predicts the residual from linear interpolation at every step.
  • Physics-informed variant (physics_weight > 0) - adds the residual of the CO2 mass balance dC/dt - [g N - lambda (C - C_out)] evaluated with central finite differences on the network output, with g, lambda and C_out learned jointly (positive by construction). This is the PINN idea from 10.3 applied as a soft constraint on a data-driven model.
  • NeuralODE - learns the right-hand side dC/dt = f_theta(C, N, time-of-day) and integrates it with a differentiable RK4 scheme (10.2). A gap is filled by integrating from its left boundary; the end mismatch is removed with a linear correction so the fill joins the right boundary.

Both imputers follow the same interface: fit(...) returns a TrainLog, and impute(y, ts, occ) returns a filled copy of y in the original units, leaving observed values unchanged.

Example:
>>> import numpy as np, pandas as pd
>>> from sensorfill.deep import BiGRUImputer
>>> ts = pd.date_range("2024-06-03", periods=600, freq="10min")
>>> y = 500 + 50 * np.sin(np.arange(600) / 20)
>>> m = BiGRUImputer(window=72, hidden=8, layers=1, device="cpu")
>>> log = m.fit(y, ts, epochs=1, steps_per_epoch=2, batch=4, gap_choices=(3,))
>>> y_gap = y.copy(); y_gap[100:103] = np.nan
>>> filled = m.impute(y_gap, ts)
>>> bool(np.isfinite(filled).all()), bool((filled[:100] == y[:100]).all())
(True, True)

 85def calendar_channels(ts) -> np.ndarray:
 86    """Encode time of day and day of week on the unit circle.
 87
 88    Args:
 89        ts: Timestamps.
 90
 91    Returns:
 92        float32 array of shape ``(n, 4)``: ``[tod_sin, tod_cos, dow_sin, dow_cos]``.
 93        The encoding is continuous at midnight and between Sunday and Monday.
 94
 95    Example:
 96        >>> calendar_channels(pd.DatetimeIndex(["2024-06-03 06:00"])).round(3)
 97        array([[1., 0., 0., 1.]], dtype=float32)
 98    """
 99    ts = pd.DatetimeIndex(ts)
100    tod = (ts.hour + ts.minute / 60.0).to_numpy() / 24.0
101    dow = ts.dayofweek.to_numpy() / 7.0
102    return np.column_stack(
103        [np.sin(2 * np.pi * tod), np.cos(2 * np.pi * tod), np.sin(2 * np.pi * dow), np.cos(2 * np.pi * dow)]
104    ).astype(np.float32)

142def _inputs(y_obs: np.ndarray, occ: np.ndarray | None, cal: np.ndarray, sc: Scaler) -> np.ndarray:
143    """Stack network input channels for one window: ``[y_lin, mask, occ, cal...]``."""
144    mask = (~np.isnan(y_obs)).astype(np.float32)
145    y_lin = (fill_linear(y_obs) - sc.y_mean) / sc.y_std
146    o = np.zeros_like(y_lin) if occ is None else (np.nan_to_num(occ) - sc.occ_mean) / sc.occ_std
147    return np.column_stack([y_lin, mask, o, cal]).astype(np.float32)

150class _GRUNet(nn.Module):
151    """Bidirectional GRU encoder with a linear read-out per time step."""
152
153    def __init__(self, n_in: int, hidden: int, layers: int, dropout: float):
154        """Build the GRU stack and the two-layer read-out head."""
155        super().__init__()
156        self.gru = nn.GRU(n_in, hidden, num_layers=layers, batch_first=True, bidirectional=True, dropout=dropout if layers > 1 else 0.0)
157        self.head = nn.Sequential(nn.Linear(2 * hidden, hidden), nn.GELU(), nn.Linear(hidden, 1))
158
159    def forward(self, x):
160        """Map ``(batch, time, channels)`` inputs to ``(batch, time)`` outputs."""
161        h, _ = self.gru(x)
162        return self.head(h).squeeze(-1)

182class BiGRUImputer:
183    """Window-based bidirectional-GRU gap filler with optional physics loss.
184
185    Args:
186        window: Window length in steps (432 = 3 days at 10 min). Gaps must be
187            shorter than ``window - 2 * context`` to be filled by one window.
188        hidden: GRU hidden size per direction.
189        layers: Number of stacked GRU layers.
190        dropout: Dropout between GRU layers.
191        physics_weight: Weight of the CO2 mass-balance residual loss (0 = off).
192        dt_hours: Grid spacing in hours (for the physics residual).
193        ode_init: Initial ``(g, lambda, C_out)`` for the learnable physics
194            parameters, e.g. from :func:`sensorfill.ode.fit_output_error`.
195        device: ``"cuda"``, ``"cpu"`` or None for automatic.
196        seed: Random seed.
197
198    Attributes:
199        net: The PyTorch network.
200        sc: Training :class:`Scaler` (None until :meth:`fit` is called).
201        log: :class:`TrainLog` of the last training run.
202    """
203
204    def __init__(
205        self,
206        window: int = 432,
207        hidden: int = 64,
208        layers: int = 2,
209        dropout: float = 0.1,
210        physics_weight: float = 0.0,
211        dt_hours: float = 1 / 6,
212        ode_init: tuple[float, float, float] = (5.0, 1.0, 420.0),
213        device: str | None = None,
214        seed: int = 0,
215    ):
216        """Create the (untrained) network and the learnable physics parameters."""
217        set_seed(seed)
218        self.window, self.dt = window, dt_hours
219        self.device = get_device(device)
220        self.net = _GRUNet(3 + 4, hidden, layers, dropout).to(self.device)
221        self.physics_weight = physics_weight
222        g, lam, cout = ode_init
223        # Log-parameterisation keeps g and lambda positive during optimisation.
224        self.log_g = nn.Parameter(torch.tensor(np.log(max(g, 1e-3)), dtype=torch.float32, device=self.device))
225        self.log_lam = nn.Parameter(torch.tensor(np.log(max(lam, 1e-3)), dtype=torch.float32, device=self.device))
226        self.c_out = nn.Parameter(torch.tensor(float(cout), dtype=torch.float32, device=self.device))
227        self.sc: Scaler | None = None
228        self.log = TrainLog()
229        self.rng = np.random.default_rng(seed)
230
231    # ------------------------------------------------------------ batching
232    def _sample_batch(self, y, occ, cal, starts, gap_choices, batch: int):
233        """Draw random windows and hide one random gap in each.
234
235        Returns tensors ``(inputs, normalised truth, hide mask, occupancy)``.
236        """
237        W = self.window
238        X, T, M, Occ = [], [], [], []
239        for s in self.rng.choice(starts, size=batch):
240            yt = y[s : s + W].copy()
241            obs = ~np.isnan(yt)
242            L = int(self.rng.choice(gap_choices))
243            try:
244                hide = block_mask(obs, L, 1, self.rng, margin=2, min_observed=0.8)
245            except ValueError:
246                hide = np.zeros(W, dtype=bool)
247            yo = yt.copy()
248            yo[hide] = np.nan
249            if np.count_nonzero(~np.isnan(yo)) < 2:
250                continue
251            X.append(_inputs(yo, None if occ is None else occ[s : s + W], cal[s : s + W], self.sc))
252            T.append(np.nan_to_num((yt - self.sc.y_mean) / self.sc.y_std))
253            M.append(hide.astype(np.float32))
254            Occ.append(np.zeros(W, np.float32) if occ is None else np.nan_to_num(occ[s : s + W]).astype(np.float32))
255
256        def to_tensor(a):
257            """Stack a list of per-window arrays into one tensor on the model device."""
258            return torch.tensor(np.stack(a), device=self.device)
259
260        return to_tensor(X), to_tensor(T).float(), to_tensor(M), to_tensor(Occ)
261
262    def _physics_residual(self, pred_norm, occ):
263        """Mean-square CO2 mass-balance residual (ppm/h) of a predicted batch."""
264        C = pred_norm * self.sc.y_std + self.sc.y_mean
265        dC = (C[:, 2:] - C[:, :-2]) / (2 * self.dt)
266        rhs = torch.exp(self.log_g) * occ[:, 1:-1] - torch.exp(self.log_lam) * (C[:, 1:-1] - self.c_out)
267        return ((dC - rhs) / self.sc.y_std).pow(2).mean()
268
269    def _forward(self, X):
270        """Normalised prediction: linear baseline (channel 0) plus learned residual."""
271        # Network predicts the residual from the (normalised) linear baseline in channel 0.
272        return X[..., 0] + self.net(X)

274    # ------------------------------------------------------------ training
275    def fit(
276        self,
277        y_train,
278        ts_train,
279        occ_train=None,
280        y_val=None,
281        ts_val=None,
282        occ_val=None,
283        epochs: int = 40,
284        steps_per_epoch: int = 100,
285        batch: int = 64,
286        lr: float = 2e-3,
287        gap_choices=(1, 3, 6, 12, 24, 48, 96, 144),
288        patience: int = 8,
289    ) -> TrainLog:
290        """Train with randomly placed artificial gaps; early-stop on validation loss.
291
292        The data loss is the mean-squared error on hidden points only. After
293        training, the weights from the epoch with the lowest validation loss
294        are restored.
295
296        Args:
297            y_train: Training target series (NaN allowed; never used as a target).
298            ts_train: Timestamps of ``y_train``.
299            occ_train: Training occupancy, or None.
300            y_val: Optional validation series for early stopping.
301            ts_val: Timestamps of ``y_val``.
302            occ_val: Validation occupancy, or None.
303            epochs: Maximum number of epochs.
304            steps_per_epoch: Mini-batches per epoch.
305            batch: Windows per mini-batch.
306            lr: Initial AdamW learning rate (cosine-annealed to 0).
307            gap_choices: Artificial gap lengths drawn during training [steps].
308            patience: Stop after this many epochs without validation improvement.
309
310        Returns:
311            The :class:`TrainLog`.
312        """
313        y_train = np.asarray(y_train, dtype=float)
314        self.sc = Scaler.fit(y_train, occ_train)
315        cal_tr = calendar_channels(ts_train)
316        W = self.window
317        starts = np.arange(0, len(y_train) - W)
318        params = list(self.net.parameters())
319        if self.physics_weight > 0:
320            params += [self.log_g, self.log_lam, self.c_out]
321        opt = torch.optim.AdamW(params, lr=lr, weight_decay=1e-4)
322        sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=epochs)
323        if y_val is not None:
324            cal_va = calendar_channels(ts_val)
325            vstarts = np.arange(0, len(y_val) - W, W // 4)
326            vrng_state = np.random.default_rng(123)
327            saved, self.rng = self.rng, vrng_state
328            Xv, Tv, Mv, Ov = self._sample_batch(np.asarray(y_val, float), occ_val, cal_va, vstarts, gap_choices, 256)
329            self.rng = saved
330        best, best_state, bad = np.inf, None, 0
331        t0 = time.perf_counter()
332        for ep in range(epochs):
333            self.net.train()
334            tot = 0.0
335            for _ in range(steps_per_epoch):
336                X, T, M, Occ = self._sample_batch(y_train, occ_train, cal_tr, starts, gap_choices, batch)
337                pred = self._forward(X)
338                loss = ((pred - T) ** 2 * M).sum() / M.sum().clamp(min=1)
339                if self.physics_weight > 0:
340                    loss = loss + self.physics_weight * self._physics_residual(pred, Occ)
341                opt.zero_grad()
342                loss.backward()
343                nn.utils.clip_grad_norm_(self.net.parameters(), 1.0)  # guards against exploding RNN gradients
344                opt.step()
345                tot += loss.item()
346            sched.step()
347            self.log.train_loss.append(tot / steps_per_epoch)
348            if y_val is not None:
349                self.net.eval()
350                with torch.no_grad():
351                    pv = self._forward(Xv)
352                    vl = (((pv - Tv) ** 2 * Mv).sum() / Mv.sum().clamp(min=1)).item()
353                self.log.val_loss.append(vl)
354                if vl < best - 1e-6:
355                    best, bad, self.log.best_epoch = vl, 0, ep
356                    best_state = {k: v.detach().clone() for k, v in self.net.state_dict().items()}
357                else:
358                    bad += 1
359                    if bad >= patience:
360                        break
361        if best_state is not None:
362            self.net.load_state_dict(best_state)
363        self.log.seconds = time.perf_counter() - t0
364        return self.log

366    @property
367    def physics_params(self) -> dict:
368        """Current learned physics parameters.
369
370        Returns:
371            ``{"g": [ppm/(person h)], "lam": [1/h], "c_out": [ppm]}``. They
372            only change during training when ``physics_weight > 0``.
373        """
374        with torch.no_grad():
375            return {"g": float(torch.exp(self.log_g)), "lam": float(torch.exp(self.log_lam)), "c_out": float(self.c_out)}

377    # ------------------------------------------------------------ inference
378    @torch.no_grad()
379    def impute(self, y, ts, occ=None) -> np.ndarray:
380        """Fill every interior gap using a window centred on it.
381
382        Gaps longer than the window minus 16 steps of context, edge gaps, and
383        every gap of a series shorter than one window fall back to linear
384        interpolation (reported, not hidden).
385
386        Args:
387            y: Series with NaNs.
388            ts: Timestamps of ``y``.
389            occ: Occupancy on the same grid, or None (must match training).
390
391        Returns:
392            Filled copy of ``y`` in the original units.
393
394        Raises:
395            RuntimeError: If :meth:`fit` has not been called.
396        """
397        if self.sc is None:
398            raise RuntimeError("call fit() first")
399        self.net.eval()
400        y = np.asarray(y, dtype=float)
401        out = fill_linear(y)
402        cal = calendar_channels(ts)
403        W, n = self.window, len(y)
404        if n < W:
405            return out
406        jobs = []
407        for a, b in gap_spans(np.isnan(y)):
408            if a == 0 or b == n or (b - a) > W - 16:
409                continue
410            s = int(np.clip((a + b) // 2 - W // 2, 0, n - W))
411            jobs.append((a, b, s))
412        for k in range(0, len(jobs), 256):
413            chunk = jobs[k : k + 256]
414            X = np.stack([_inputs(y[s : s + W], None if occ is None else occ[s : s + W], cal[s : s + W], self.sc) for _, _, s in chunk])
415            pred = self._forward(torch.tensor(X, device=self.device)).cpu().numpy() * self.sc.y_std + self.sc.y_mean
416            for (a, b, s), p in zip(chunk, pred, strict=True):
417                out[a:b] = p[a - s : b - s]
418        return out

428    def flops_per_window(self) -> int:
429        """Approximate forward-pass FLOPs for one window (multiply-add = 2 FLOPs).
430
431        A GRU layer costs ``3 * 2 * (h*i + h*h)`` multiply-adds per step and
432        direction; the head adds ``2*(2h*h + h)``.
433
434        Returns:
435            FLOPs for one forward pass over one window.
436        """
437        h, W, flops, i = self.net.gru.hidden_size, self.window, 0, self.net.gru.input_size
438        for layer in range(self.net.gru.num_layers):
439            inp = i if layer == 0 else 2 * h
440            flops += 2 * 2 * 3 * (h * inp + h * h) * W  # 2 directions, 3 gates, 2 flops per MAC
441        flops += 2 * (2 * h * h + h) * W
442        return int(flops)

446class _ODEFunc(nn.Module):
447    """MLP right-hand side ``f(C, N, tod)`` returning dC/dt in normalised units per hour."""
448
449    def __init__(self, hidden: int = 64):
450        """Two hidden tanh layers (smooth, so RK4 sees a smooth vector field)."""
451        super().__init__()
452        self.net = nn.Sequential(nn.Linear(4, hidden), nn.Tanh(), nn.Linear(hidden, hidden), nn.Tanh(), nn.Linear(hidden, 1))
453
454    def forward(self, c, u):
455        """dC/dt for states ``c`` (B,) and inputs ``u`` (B, 3) = [occ, tod_sin, tod_cos]."""
456        return self.net(torch.cat([c[:, None], u], dim=1)).squeeze(-1)

459def rk4_rollout(func, c0, U, dt):
460    """Differentiable classical RK4 over a sequence of inputs.
461
462    Inputs are held at their left-endpoint value for the half step and use the
463    next sample for the full step (linear interpolation at the midpoint).
464
465    Args:
466        func: Module ``f(c, u)``.
467        c0: Initial states, shape ``(B,)``.
468        U: Exogenous inputs, shape ``(B, T, k)``.
469        dt: Step in hours.
470
471    Returns:
472        States, shape ``(B, T)`` with ``out[:, 0] = c0``.
473
474    Example:
475        >>> decay = lambda c, u: -c  # dc/dt = -c
476        >>> out = rk4_rollout(decay, torch.ones(1), torch.zeros(1, 11, 1), dt=0.1)
477        >>> round(float(out[0, -1]), 5)  # e^-1
478        0.36788
479    """
480    cs = [c0]
481    c = c0
482    for k in range(U.shape[1] - 1):
483        u0, u1 = U[:, k], U[:, k + 1]
484        um = 0.5 * (u0 + u1)
485        k1 = func(c, u0)
486        k2 = func(c + 0.5 * dt * k1, um)
487        k3 = func(c + 0.5 * dt * k2, um)
488        k4 = func(c + dt * k3, u1)
489        c = c + dt / 6 * (k1 + 2 * k2 + 2 * k3 + k4)
490        cs.append(c)
491    return torch.stack(cs, dim=1)

233    def _sample_batch(self, y, occ, cal, starts, gap_choices, batch: int):
234        """Draw random windows and hide one random gap in each.
235
236        Returns tensors ``(inputs, normalised truth, hide mask, occupancy)``.
237        """
238        W = self.window
239        X, T, M, Occ = [], [], [], []
240        for s in self.rng.choice(starts, size=batch):
241            yt = y[s : s + W].copy()
242            obs = ~np.isnan(yt)
243            L = int(self.rng.choice(gap_choices))
244            try:
245                hide = block_mask(obs, L, 1, self.rng, margin=2, min_observed=0.8)
246            except ValueError:

248            yo = yt.copy()

250            if np.count_nonzero(~np.isnan(yo)) < 2:
251                continue
252            X.append(_inputs(yo, None if occ is None else occ[s : s + W], cal[s : s + W], self.sc))
253            T.append(np.nan_to_num((yt - self.sc.y_mean) / self.sc.y_std))
254            M.append(hide.astype(np.float32))
255            Occ.append(np.zeros(W, np.float32) if occ is None else np.nan_to_num(occ[s : s + W]).astype(np.float32))

258            """Stack a list of per-window arrays into one tensor on the model device."""
259            return torch.tensor(np.stack(a), device=self.device)

263    def _physics_residual(self, pred_norm, occ):
264        """Mean-square CO2 mass-balance residual (ppm/h) of a predicted batch."""
265        C = pred_norm * self.sc.y_std + self.sc.y_mean
266        dC = (C[:, 2:] - C[:, :-2]) / (2 * self.dt)
267        rhs = torch.exp(self.log_g) * occ[:, 1:-1] - torch.exp(self.log_lam) * (C[:, 1:-1] - self.c_out)
268        return ((dC - rhs) / self.sc.y_std).pow(2).mean()

272        # Network predicts the residual from the (normalised) linear baseline in channel 0.

429    def flops_per_window(self) -> int:
430        """Approximate forward-pass FLOPs for one window (multiply-add = 2 FLOPs).
431
432        A GRU layer costs ``3 * 2 * (h*i + h*h)`` multiply-adds per step and
433        direction; the head adds ``2*(2h*h + h)``.
434
435        Returns:
436            FLOPs for one forward pass over one window.
437        """
438        h, W, flops, i = self.net.gru.hidden_size, self.window, 0, self.net.gru.input_size
439        for layer in range(self.net.gru.num_layers):
440            inp = i if layer == 0 else 2 * h
441            flops += 2 * 2 * 3 * (h * inp + h * h) * W  # 2 directions, 3 gates, 2 flops per MAC
442        flops += 2 * (2 * h * h + h) * W
443        return int(flops)

460def rk4_rollout(func, c0, U, dt):
461    """Differentiable classical RK4 over a sequence of inputs.
462
463    Inputs are held at their left-endpoint value for the half step and use the
464    next sample for the full step (linear interpolation at the midpoint).
465
466    Args:
467        func: Module ``f(c, u)``.
468        c0: Initial states, shape ``(B,)``.
469        U: Exogenous inputs, shape ``(B, T, k)``.
470        dt: Step in hours.
471
472    Returns:
473        States, shape ``(B, T)`` with ``out[:, 0] = c0``.
474
475    Example:
476        >>> decay = lambda c, u: -c  # dc/dt = -c
477        >>> out = rk4_rollout(decay, torch.ones(1), torch.zeros(1, 11, 1), dt=0.1)
478        >>> round(float(out[0, -1]), 5)  # e^-1
479        0.36788
480    """
481    cs = [c0]
482    c = c0
483    for k in range(U.shape[1] - 1):
484        u0, u1 = U[:, k], U[:, k + 1]
485        um = 0.5 * (u0 + u1)
486        k1 = func(c, u0)
487        k2 = func(c + 0.5 * dt * k1, um)
488        k3 = func(c + 0.5 * dt * k2, um)
489        k4 = func(c + dt * k3, u1)
490        c = c + dt / 6 * (k1 + 2 * k2 + 2 * k3 + k4)
491        cs.append(c)
492    return torch.stack(cs, dim=1)
