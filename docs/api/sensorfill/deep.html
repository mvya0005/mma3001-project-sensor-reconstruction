sensorfill.deep

Deep-learning imputers in PyTorch (unit topics 5.4, 10.2, 10.3).

Designed to train on a Colab GPU (device="cuda") but runs on CPU too.

  • BiGRUImputer - bidirectional GRU that reads a window around each gap (observed values, the mask, occupancy and calendar channels) and predicts the residual from linear interpolation at every step.
  • Physics-informed variant (physics_weight > 0) - adds the residual of the CO2 mass balance dC/dt - [g N - lambda (C - C_out)] evaluated with central finite differences on the network output, with g, lambda and C_out learned jointly (positive by construction). This is the PINN idea from 10.3 applied as a soft constraint on a data-driven model.
  • NeuralODE - learns the right-hand side dC/dt = f_theta(C, N, time-of-day) and integrates it with a differentiable RK4 scheme (10.2). A gap is filled by integrating from its left boundary; the end mismatch is removed with a linear correction so the fill joins the right boundary.

 52def calendar_channels(ts) -> np.ndarray:
 53    """Time-of-day and day-of-week encoded as sin/cos, shape ``(n, 4)``."""
 54    ts = pd.DatetimeIndex(ts)
 55    tod = (ts.hour + ts.minute / 60.0).to_numpy() / 24.0
 56    dow = ts.dayofweek.to_numpy() / 7.0
 57    return np.column_stack(
 58        [np.sin(2 * np.pi * tod), np.cos(2 * np.pi * tod), np.sin(2 * np.pi * dow), np.cos(2 * np.pi * dow)]
 59    ).astype(np.float32)

 71    @classmethod
 72    def fit(cls, y, occ=None) -> "Scaler":
 73        """Estimate mean and standard deviation (ignoring NaN)."""
 74        o = np.zeros(1) if occ is None else np.asarray(occ, dtype=float)

 78def _inputs(y_obs: np.ndarray, occ: np.ndarray | None, cal: np.ndarray, sc: Scaler) -> np.ndarray:
 79    """Stack network input channels for one window: ``[y_lin, mask, occ, cal...]``."""
 80    mask = (~np.isnan(y_obs)).astype(np.float32)
 81    y_lin = (fill_linear(y_obs) - sc.y_mean) / sc.y_std
 82    o = np.zeros_like(y_lin) if occ is None else (np.nan_to_num(occ) - sc.occ_mean) / sc.occ_std
 83    return np.column_stack([y_lin, mask, o, cal]).astype(np.float32)

 86class _GRUNet(nn.Module):
 87    """Bidirectional GRU encoder with a linear read-out per time step."""
 88
 89    def __init__(self, n_in: int, hidden: int, layers: int, dropout: float):
 90        super().__init__()
 91        self.gru = nn.GRU(n_in, hidden, num_layers=layers, batch_first=True, bidirectional=True, dropout=dropout if layers > 1 else 0.0)
 92        self.head = nn.Sequential(nn.Linear(2 * hidden, hidden), nn.GELU(), nn.Linear(hidden, 1))
 93
 94    def forward(self, x):  # x: (batch, time, channels)
 95        h, _ = self.gru(x)
 96        return self.head(h).squeeze(-1)

109class BiGRUImputer:
110    """Window-based bidirectional-GRU gap filler with optional physics loss.
111
112    Args:
113        window: Window length in steps (432 = 3 days at 10 min). Gaps must be
114            shorter than ``window - 2 * context`` to be filled by one window.
115        hidden: GRU hidden size per direction.
116        layers: Number of stacked GRU layers.
117        dropout: Dropout between GRU layers.
118        physics_weight: Weight of the CO2 mass-balance residual loss (0 = off).
119        dt_hours: Grid spacing in hours (for the physics residual).
120        ode_init: Initial ``(g, lambda, C_out)`` for the learnable physics
121            parameters, e.g. from :func:`sensorfill.ode.fit_output_error`.
122        device: ``"cuda"``, ``"cpu"`` or None for automatic.
123        seed: Random seed.
124    """
125
126    def __init__(
127        self,
128        window: int = 432,
129        hidden: int = 64,
130        layers: int = 2,
131        dropout: float = 0.1,
132        physics_weight: float = 0.0,
133        dt_hours: float = 1 / 6,
134        ode_init: tuple[float, float, float] = (5.0, 1.0, 420.0),
135        device: str | None = None,
136        seed: int = 0,
137    ):
138        set_seed(seed)
139        self.window, self.dt = window, dt_hours
140        self.device = get_device(device)
141        self.net = _GRUNet(3 + 4, hidden, layers, dropout).to(self.device)
142        self.physics_weight = physics_weight
143        g, lam, cout = ode_init
144        # Log-parameterisation keeps g and lambda positive during optimisation.
145        self.log_g = nn.Parameter(torch.tensor(np.log(max(g, 1e-3)), dtype=torch.float32, device=self.device))
146        self.log_lam = nn.Parameter(torch.tensor(np.log(max(lam, 1e-3)), dtype=torch.float32, device=self.device))
147        self.c_out = nn.Parameter(torch.tensor(float(cout), dtype=torch.float32, device=self.device))
148        self.sc: Scaler | None = None
149        self.log = TrainLog()
150        self.rng = np.random.default_rng(seed)
151
152    # ------------------------------------------------------------ batching
153    def _sample_batch(self, y, occ, cal, starts, gap_choices, batch: int):
154        W = self.window
155        X, T, M, O = [], [], [], []
156        for s in self.rng.choice(starts, size=batch):
157            yt = y[s : s + W].copy()
158            obs = ~np.isnan(yt)
159            L = int(self.rng.choice(gap_choices))
160            try:
161                hide = block_mask(obs, L, 1, self.rng, margin=2, min_observed=0.8)
162            except ValueError:
163                hide = np.zeros(W, dtype=bool)
164            yo = yt.copy()
165            yo[hide] = np.nan
166            if np.count_nonzero(~np.isnan(yo)) < 2:
167                continue
168            X.append(_inputs(yo, None if occ is None else occ[s : s + W], cal[s : s + W], self.sc))
169            T.append(np.nan_to_num((yt - self.sc.y_mean) / self.sc.y_std))
170            M.append(hide.astype(np.float32))
171            O.append(np.zeros(W, np.float32) if occ is None else np.nan_to_num(occ[s : s + W]).astype(np.float32))
172        f = lambda a: torch.tensor(np.stack(a), device=self.device)
173        return f(X), f(T).float(), f(M), f(O)
174
175    def _physics_residual(self, pred_norm, occ):
176        """Mean-square CO2 mass-balance residual (ppm/h) of a predicted batch."""
177        C = pred_norm * self.sc.y_std + self.sc.y_mean
178        dC = (C[:, 2:] - C[:, :-2]) / (2 * self.dt)
179        rhs = torch.exp(self.log_g) * occ[:, 1:-1] - torch.exp(self.log_lam) * (C[:, 1:-1] - self.c_out)
180        return ((dC - rhs) / self.sc.y_std).pow(2).mean()
181
182    def _forward(self, X):
183        # Network predicts the residual from the (normalised) linear baseline in channel 0.
184        return X[..., 0] + self.net(X)
185
186    # ------------------------------------------------------------ training
187    def fit(
188        self,
189        y_train,
190        ts_train,
191        occ_train=None,
192        y_val=None,
193        ts_val=None,
194        occ_val=None,
195        epochs: int = 40,
196        steps_per_epoch: int = 100,
197        batch: int = 64,
198        lr: float = 2e-3,
199        gap_choices=(1, 3, 6, 12, 24, 48, 96, 144),
200        patience: int = 8,
201    ) -> TrainLog:
202        """Train with randomly placed artificial gaps; early-stop on validation loss.
203
204        The data loss is the mean-squared error on hidden points only. Returns
205        the :class:`TrainLog` and restores the best-validation weights.
206        """
207        y_train = np.asarray(y_train, dtype=float)
208        self.sc = Scaler.fit(y_train, occ_train)
209        cal_tr = calendar_channels(ts_train)
210        W = self.window
211        starts = np.arange(0, len(y_train) - W)
212        params = list(self.net.parameters())
213        if self.physics_weight > 0:
214            params += [self.log_g, self.log_lam, self.c_out]
215        opt = torch.optim.AdamW(params, lr=lr, weight_decay=1e-4)
216        sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=epochs)
217        if y_val is not None:
218            cal_va = calendar_channels(ts_val)
219            vstarts = np.arange(0, len(y_val) - W, W // 4)
220            vrng_state = np.random.default_rng(123)
221            saved, self.rng = self.rng, vrng_state
222            Xv, Tv, Mv, Ov = self._sample_batch(np.asarray(y_val, float), occ_val, cal_va, vstarts, gap_choices, 256)
223            self.rng = saved
224        best, best_state, bad = np.inf, None, 0
225        t0 = time.perf_counter()
226        for ep in range(epochs):
227            self.net.train()
228            tot = 0.0
229            for _ in range(steps_per_epoch):
230                X, T, M, O = self._sample_batch(y_train, occ_train, cal_tr, starts, gap_choices, batch)
231                pred = self._forward(X)
232                loss = ((pred - T) ** 2 * M).sum() / M.sum().clamp(min=1)
233                if self.physics_weight > 0:
234                    loss = loss + self.physics_weight * self._physics_residual(pred, O)
235                opt.zero_grad()
236                loss.backward()
237                nn.utils.clip_grad_norm_(self.net.parameters(), 1.0)  # guards against exploding RNN gradients
238                opt.step()
239                tot += loss.item()
240            sched.step()
241            self.log.train_loss.append(tot / steps_per_epoch)
242            if y_val is not None:
243                self.net.eval()
244                with torch.no_grad():
245                    pv = self._forward(Xv)
246                    vl = (((pv - Tv) ** 2 * Mv).sum() / Mv.sum()).item()
247                self.log.val_loss.append(vl)
248                if vl < best - 1e-6:
249                    best, bad, self.log.best_epoch = vl, 0, ep
250                    best_state = {k: v.detach().clone() for k, v in self.net.state_dict().items()}
251                else:
252                    bad += 1
253                    if bad >= patience:
254                        break
255        if best_state is not None:
256            self.net.load_state_dict(best_state)
257        self.log.seconds = time.perf_counter() - t0
258        return self.log

260    @property
261    def physics_params(self) -> dict:
262        """Current learned ``g`` [ppm/(person h)], ``lambda`` [1/h] and ``C_out`` [ppm]."""
263        with torch.no_grad():
264            return {"g": float(torch.exp(self.log_g)), "lam": float(torch.exp(self.log_lam)), "c_out": float(self.c_out)}

266    # ------------------------------------------------------------ inference
267    @torch.no_grad()
268    def impute(self, y, ts, occ=None) -> np.ndarray:
269        """Fill every interior gap using a window centred on it.
270
271        Gaps longer than the window minus 16 steps of context, and edge gaps,
272        fall back to linear interpolation (reported, not hidden).
273        """
274        if self.sc is None:
275            raise RuntimeError("call fit() first")
276        self.net.eval()
277        y = np.asarray(y, dtype=float)
278        out = fill_linear(y)
279        cal = calendar_channels(ts)
280        W, n = self.window, len(y)
281        jobs = []
282        for a, b in gap_spans(np.isnan(y)):
283            if a == 0 or b == n or (b - a) > W - 16:
284                continue
285            s = int(np.clip((a + b) // 2 - W // 2, 0, n - W))
286            jobs.append((a, b, s))
287        for k in range(0, len(jobs), 256):
288            chunk = jobs[k : k + 256]
289            X = np.stack([_inputs(y[s : s + W], None if occ is None else occ[s : s + W], cal[s : s + W], self.sc) for _, _, s in chunk])
290            pred = self._forward(torch.tensor(X, device=self.device)).cpu().numpy() * self.sc.y_std + self.sc.y_mean
291            for (a, b, s), p in zip(chunk, pred):
292                out[a:b] = p[a - s : b - s]
293        return out

295    def n_parameters(self) -> int:
296        """Number of trainable network weights."""
297        return sum(p.numel() for p in self.net.parameters())

299    def flops_per_window(self) -> int:
300        """Approximate forward-pass FLOPs for one window (multiply-add = 2 FLOPs).
301
302        A GRU layer costs ``3 * 2 * (h*i + h*h)`` multiply-adds per step and
303        direction; the head adds ``2*(2h*h + h)``.
304        """
305        h, W, flops, i = self.net.gru.hidden_size, self.window, 0, self.net.gru.input_size
306        for layer in range(self.net.gru.num_layers):
307            inp = i if layer == 0 else 2 * h
308            flops += 2 * 2 * 3 * (h * inp + h * h) * W  # 2 directions, 3 gates, 2 flops per MAC
309        flops += 2 * (2 * h * h + h) * W
310        return int(flops)

314class _ODEFunc(nn.Module):
315    """MLP right-hand side ``f(C, N, tod)`` returning dC/dt in normalised units per hour."""
316
317    def __init__(self, hidden: int = 64):
318        super().__init__()
319        self.net = nn.Sequential(nn.Linear(4, hidden), nn.Tanh(), nn.Linear(hidden, hidden), nn.Tanh(), nn.Linear(hidden, 1))
320
321    def forward(self, c, u):  # c: (B,), u: (B, 3) = [occ, tod_sin, tod_cos]
322        return self.net(torch.cat([c[:, None], u], dim=1)).squeeze(-1)

325def rk4_rollout(func, c0, U, dt):
326    """Differentiable classical RK4 over a sequence of inputs.
327
328    Inputs are held at their left-endpoint value for the half step and use the
329    next sample for the full step (linear interpolation at the midpoint).
330
331    Args:
332        func: Module ``f(c, u)``.
333        c0: Initial states, shape ``(B,)``.
334        U: Exogenous inputs, shape ``(B, T, k)``.
335        dt: Step in hours.
336
337    Returns:
338        States, shape ``(B, T)`` with ``out[:, 0] = c0``.
339    """
340    cs = [c0]
341    c = c0
342    for k in range(U.shape[1] - 1):
343        u0, u1 = U[:, k], U[:, k + 1]
344        um = 0.5 * (u0 + u1)
345        k1 = func(c, u0)
346        k2 = func(c + 0.5 * dt * k1, um)
347        k3 = func(c + 0.5 * dt * k2, um)
348        k4 = func(c + dt * k3, u1)
349        c = c + dt / 6 * (k1 + 2 * k2 + 2 * k3 + k4)
350        cs.append(c)
351    return torch.stack(cs, dim=1)

354class NeuralODE:
355    """Learned CO2 dynamics integrated with RK4.
356
357    Args:
358        horizon: Training segment length in steps (36 = 6 h).
359        hidden: Width of the MLP.
360        dt_hours: Grid spacing.
361        device: Torch device or None.
362        seed: Random seed.
363    """
364
365    def __init__(self, horizon: int = 36, hidden: int = 64, dt_hours: float = 1 / 6, device=None, seed: int = 0):
366        set_seed(seed)
367        self.h, self.dt = horizon, dt_hours
368        self.device = get_device(device)
369        self.func = _ODEFunc(hidden).to(self.device)
370        self.sc: Scaler | None = None
371        self.log = TrainLog()
372        self.rng = np.random.default_rng(seed)
373
374    def _U(self, occ, ts):
375        cal = calendar_channels(ts)[:, :2]
376        o = (np.nan_to_num(occ) - self.sc.occ_mean) / self.sc.occ_std
377        return np.column_stack([o, cal]).astype(np.float32)
378
379    def fit(self, y, ts, occ, epochs: int = 30, steps_per_epoch: int = 100, batch: int = 128, lr: float = 3e-3) -> TrainLog:
380        """Fit by multiple shooting on random fully observed segments of length ``horizon``."""
381        y = np.asarray(y, dtype=float)
382        self.sc = Scaler.fit(y, occ)
383        yn = (y - self.sc.y_mean) / self.sc.y_std
384        U = self._U(occ, ts)
385        ok = ~np.isnan(yn)
386        c = np.concatenate([[0], np.cumsum(ok)])
387        starts = np.flatnonzero((c[self.h :] - c[: -self.h]) == self.h)
388        if starts.size == 0:
389            raise ValueError("no fully observed segment of the requested horizon")
390        opt = torch.optim.Adam(self.func.parameters(), lr=lr)
391        t0 = time.perf_counter()
392        for _ in range(epochs):
393            tot = 0.0
394            for _ in range(steps_per_epoch):
395                s = self.rng.choice(starts, size=batch)
396                idx = s[:, None] + np.arange(self.h)[None, :]
397                Y = torch.tensor(yn[idx], dtype=torch.float32, device=self.device)
398                Ub = torch.tensor(U[idx], device=self.device)
399                pred = rk4_rollout(self.func, Y[:, 0], Ub, self.dt)
400                loss = ((pred - Y) ** 2).mean()
401                opt.zero_grad()
402                loss.backward()
403                opt.step()
404                tot += loss.item()
405            self.log.train_loss.append(tot / steps_per_epoch)
406        self.log.seconds = time.perf_counter() - t0
407        return self.log
408
409    @torch.no_grad()
410    def impute(self, y, ts, occ, correct_end: bool = True) -> np.ndarray:
411        """Integrate across each interior gap from its left boundary.
412
413        Args:
414            y: Series with NaNs.
415            ts: Timestamps.
416            occ: Occupancy on the grid.
417            correct_end: Remove the mismatch with the right boundary by adding a
418                linearly growing correction (a cheap two-point BVP fix).
419
420        Returns:
421            Filled copy (edge gaps use linear/hold).
422        """
423        y = np.asarray(y, dtype=float)
424        out = fill_linear(y)
425        U = self._U(occ, ts)
426        for a, b in gap_spans(np.isnan(y)):
427            if a == 0 or b == len(y):
428                continue
429            seg = slice(a - 1, b + 1)
430            c0 = torch.tensor([(y[a - 1] - self.sc.y_mean) / self.sc.y_std], dtype=torch.float32, device=self.device)
431            Ub = torch.tensor(U[seg][None], device=self.device)
432            traj = rk4_rollout(self.func, c0, Ub, self.dt)[0].cpu().numpy() * self.sc.y_std + self.sc.y_mean
433            if correct_end:
434                traj = traj + (y[b] - traj[-1]) * np.linspace(0, 1, traj.size)
435            out[a:b] = traj[1:-1]
436        return out

154    def _sample_batch(self, y, occ, cal, starts, gap_choices, batch: int):
155        W = self.window
156        X, T, M, O = [], [], [], []
157        for s in self.rng.choice(starts, size=batch):
158            yt = y[s : s + W].copy()
159            obs = ~np.isnan(yt)
160            L = int(self.rng.choice(gap_choices))
161            try:
162                hide = block_mask(obs, L, 1, self.rng, margin=2, min_observed=0.8)
163            except ValueError:
164                hide = np.zeros(W, dtype=bool)
165            yo = yt.copy()
166            yo[hide] = np.nan
167            if np.count_nonzero(~np.isnan(yo)) < 2:
168                continue
169            X.append(_inputs(yo, None if occ is None else occ[s : s + W], cal[s : s + W], self.sc))
170            T.append(np.nan_to_num((yt - self.sc.y_mean) / self.sc.y_std))
171            M.append(hide.astype(np.float32))
172            O.append(np.zeros(W, np.float32) if occ is None else np.nan_to_num(occ[s : s + W]).astype(np.float32))
173        f = lambda a: torch.tensor(np.stack(a), device=self.device)
174        return f(X), f(T).float(), f(M), f(O)

176    def _physics_residual(self, pred_norm, occ):
177        """Mean-square CO2 mass-balance residual (ppm/h) of a predicted batch."""
178        C = pred_norm * self.sc.y_std + self.sc.y_mean
179        dC = (C[:, 2:] - C[:, :-2]) / (2 * self.dt)
180        rhs = torch.exp(self.log_g) * occ[:, 1:-1] - torch.exp(self.log_lam) * (C[:, 1:-1] - self.c_out)
181        return ((dC - rhs) / self.sc.y_std).pow(2).mean()

def rk4_rollout(func, c0, U, dt): 
326def rk4_rollout(func, c0, U, dt):
327    """Differentiable classical RK4 over a sequence of inputs.
328
329    Inputs are held at their left-endpoint value for the half step and use the
330    next sample for the full step (linear interpolation at the midpoint).
331
332    Args:
333        func: Module ``f(c, u)``.
334        c0: Initial states, shape ``(B,)``.
335        U: Exogenous inputs, shape ``(B, T, k)``.
336        dt: Step in hours.
337
338    Returns:
339        States, shape ``(B, T)`` with ``out[:, 0] = c0``.
340    """
341    cs = [c0]
342    c = c0
343    for k in range(U.shape[1] - 1):
344        u0, u1 = U[:, k], U[:, k + 1]
345        um = 0.5 * (u0 + u1)
346        k1 = func(c, u0)
347        k2 = func(c + 0.5 * dt * k1, um)
348        k3 = func(c + 0.5 * dt * k2, um)
349        k4 = func(c + dt * k3, u1)
350        c = c + dt / 6 * (k1 + 2 * k2 + 2 * k3 + k4)
351        cs.append(c)
352    return torch.stack(cs, dim=1)
