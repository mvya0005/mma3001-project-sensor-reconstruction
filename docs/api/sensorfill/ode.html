sensorfill.ode

Physics-based (grey-box) CO2 reconstruction with ordinary differential equations.

Unit topics 2.1 (optimisation), 7.1-7.5 (finite differences, IVPs, Euler, Runge-Kutta), 9.1-9.4 (stability, stiffness, implicit methods, shooting).

Model

A well-mixed zone of volume V with N(t) occupants each emitting CO2 at G and an outdoor-air supply Q obeys the mass balance

$$V \frac{dC}{dt} = G N(t) - Q (C - C_{out})$$

Dividing by V gives the linear, first-order ODE used here

$$\frac{dC}{dt} = g N(t) - \lambda (C - C_{out}) + s$$

with lumped parameters g = G/V [ppm / (person h)], air-change rate lambda = Q/V [1/h], background C_out [ppm], and an optional constant source s [ppm/h] that absorbs model error. Time is in hours.

Assumptions (all stated in the report): perfect mixing; constant ventilation rate; occupancy measured in another zone is a proportional proxy for this zone (so g is an effective, not physical, emission rate); outdoor CO2 constant over the study period.

Example:
>>> import numpy as np
>>> from sensorfill.ode import CO2Params, simulate_co2
>>> p = CO2Params(g=0.0, lam=1.0, c_out=420.0)  # empty room, 1 air change per hour
>>> c = simulate_co2(p, np.array([0.0, 1.0]), np.zeros(2), c0=800.0, method="exact_linear")
>>> round(float(c[-1]), 1)  # 420 + 380 e^-1
559.8

 44# --------------------------------------------------------- finite differences
 45def finite_difference(y: np.ndarray, h: float, scheme: str = "central") -> np.ndarray:
 46    """First derivative of uniformly sampled data.
 47
 48    Args:
 49        y: Samples (NaN propagates).
 50        h: Sample spacing.
 51        scheme: ``"forward"`` (O(h)), ``"backward"`` (O(h)) or ``"central"`` (O(h^2)).
 52            End points use the one-sided first-order formula.
 53
 54    Returns:
 55        Derivative estimate, same length as ``y``.
 56
 57    Raises:
 58        ValueError: If ``scheme`` is not one of the three names.
 59
 60    Example:
 61        >>> finite_difference([0.0, 1.0, 4.0, 9.0], h=1.0)  # y = x^2
 62        array([1., 2., 4., 5.])
 63    """
 64    y = np.asarray(y, dtype=float)
 65    d = np.empty_like(y)
 66    if scheme == "forward":
 67        d[:-1] = (y[1:] - y[:-1]) / h
 68        d[-1] = d[-2]
 69    elif scheme == "backward":
 70        d[1:] = (y[1:] - y[:-1]) / h
 71        d[0] = d[1]
 72    elif scheme == "central":
 73        d[1:-1] = (y[2:] - y[:-2]) / (2 * h)
 74        d[0] = (y[1] - y[0]) / h
 75        d[-1] = (y[-1] - y[-2]) / h
 76    else:
 77        raise ValueError("scheme must be forward, backward or central")
 78    return d

 81def second_difference(y: np.ndarray, h: float) -> np.ndarray:
 82    """Central second derivative ``(y[i+1] - 2y[i] + y[i-1]) / h^2``.
 83
 84    Args:
 85        y: Uniform samples.
 86        h: Sample spacing.
 87
 88    Returns:
 89        Second-derivative estimate, same length as ``y``; the two end values
 90        are NaN because the stencil does not fit there.
 91
 92    Example:
 93        >>> second_difference([0.0, 1.0, 4.0, 9.0], h=1.0)
 94        array([nan,  2.,  2., nan])
 95    """
 96    y = np.asarray(y, dtype=float)
 97    d = np.full_like(y, np.nan)
 98    d[1:-1] = (y[2:] - 2 * y[1:-1] + y[:-2]) / h**2
 99    return d

102# ------------------------------------------------------------ IVP integrators
103def integrate(
104    f: Callable[[float, np.ndarray], np.ndarray],
105    y0,
106    t: np.ndarray,
107    method: str = "rk4",
108    newton_tol: float = 1e-10,
109    newton_max: int = 50,
110) -> np.ndarray:
111    """Integrate ``y' = f(t, y)`` on the given time grid with a fixed-step method.
112
113    Args:
114        f: Right-hand side; takes a float time and a 1-D state, returns an array.
115        y0: Initial state (scalar or 1-D).
116        t: Increasing time grid; the step may vary.
117        method: ``"euler"`` (explicit, order 1), ``"heun"`` (explicit RK2),
118            ``"midpoint"`` (explicit RK2), ``"rk4"`` (classical RK, order 4),
119            ``"implicit_euler"`` (backward Euler, order 1, A-stable) or
120            ``"trapezoid"`` (implicit, order 2, A-stable).
121        newton_tol: Relative step-size tolerance of the Newton iteration used
122            by the implicit methods (Jacobian by forward finite differences).
123        newton_max: Maximum Newton iterations per implicit step.
124
125    Returns:
126        Array of shape ``(len(t), dim)``.
127
128    Raises:
129        RuntimeError: If the implicit Newton iteration fails to converge.
130        ValueError: If ``method`` is unknown.
131
132    Example:
133        >>> t = np.linspace(0.0, 1.0, 11)
134        >>> Y = integrate(lambda t, y: -y, 1.0, t, method="rk4")  # exact: e^-1
135        >>> round(float(Y[-1, 0]), 6)
136        0.36788
137    """
138    y0 = np.atleast_1d(np.asarray(y0, dtype=float))
139    Y = np.empty((len(t), y0.size))
140    Y[0] = y0
141    for n in range(len(t) - 1):
142        h, tn, yn = t[n + 1] - t[n], t[n], Y[n]
143        if method == "euler":
144            Y[n + 1] = yn + h * f(tn, yn)
145        elif method == "heun":
146            k1 = f(tn, yn)
147            k2 = f(tn + h, yn + h * k1)
148            Y[n + 1] = yn + h / 2 * (k1 + k2)
149        elif method == "midpoint":
150            k1 = f(tn, yn)
151            Y[n + 1] = yn + h * f(tn + h / 2, yn + h / 2 * k1)
152        elif method == "rk4":
153            k1 = f(tn, yn)
154            k2 = f(tn + h / 2, yn + h / 2 * k1)
155            k3 = f(tn + h / 2, yn + h / 2 * k2)
156            k4 = f(tn + h, yn + h * k3)
157            Y[n + 1] = yn + h / 6 * (k1 + 2 * k2 + 2 * k3 + k4)
158        elif method in ("implicit_euler", "trapezoid"):
159            theta = 1.0 if method == "implicit_euler" else 0.5
160            fn = f(tn, yn)
161
162            def G(z, yn=yn, h=h, tn=tn, fn=fn, theta=theta):
163                """Residual of the implicit (theta-method) update; zero at the new state."""
164                return z - yn - h * (theta * f(tn + h, z) + (1 - theta) * fn)
165
166            z = yn + h * fn  # explicit predictor
167            for _ in range(newton_max):
168                g = G(z)
169                eps = 1e-7 * (1 + np.abs(z))
170                J = np.column_stack([(G(z + e) - g) / e[i] for i, e in enumerate(np.diag(eps))])
171                dz = np.linalg.solve(J, -g)
172                z = z + dz
173                if np.max(np.abs(dz)) < newton_tol * (1 + np.max(np.abs(z))):
174                    break
175            else:
176                raise RuntimeError("Newton iteration did not converge in implicit step")
177            Y[n + 1] = z
178        else:
179            raise ValueError(f"unknown method {method!r}")
180    return Y

187def amplification_factor(method: str, z: complex) -> complex:
188    """Stability function ``R(z)`` of a method for the test equation ``y' = lambda y``.
189
190    One step multiplies the solution by ``R(z)`` with ``z = h lambda``, so a
191    step is stable when ``|R(z)| <= 1``. For the CO2 model ``z = -h lambda``.
192
193    Args:
194        method: A key of :data:`ORDER`.
195        z: ``h * lambda`` (complex allowed).
196
197    Returns:
198        ``R(z)``.
199
200    Raises:
201        ValueError: If ``method`` is unknown.
202
203    Example:
204        >>> amplification_factor("euler", -3.0), amplification_factor("implicit_euler", -3.0)
205        (-2.0, 0.25)
206    """
207    if method == "euler":
208        return 1 + z
209    if method in ("heun", "midpoint"):
210        return 1 + z + z**2 / 2
211    if method == "rk4":
212        return 1 + z + z**2 / 2 + z**3 / 6 + z**4 / 24
213    if method == "implicit_euler":
214        return 1 / (1 - z)
215    if method == "trapezoid":
216        return (1 + z / 2) / (1 - z / 2)
217    raise ValueError(method)

220# ----------------------------------------------------------- CO2 grey box
221@dataclass
222class CO2Params:
223    """Parameters of the lumped CO2 mass balance (time unit: hours).
224
225    Attributes:
226        g: Effective generation per occupant [ppm / (person h)].
227        lam: Air-change rate ``Q/V`` [1/h].
228        c_out: Background (outdoor/supply) concentration [ppm].
229        s: Constant unmodelled source [ppm/h].
230    """
231
232    g: float
233    lam: float
234    c_out: float
235    s: float = 0.0
236
237    @property
238    def time_constant_h(self) -> float:
239        """Room time constant ``1/lambda`` in hours.
240
241        Returns:
242            The e-folding time of a CO2 disturbance once the room empties.
243
244        Example:
245            >>> CO2Params(g=10.0, lam=2.0, c_out=420.0).time_constant_h
246            0.5
247        """
248        return 1.0 / self.lam

251def co2_rhs(params: CO2Params, occ: Callable[[float], float]):
252    """Build the right-hand side of the CO2 mass balance.
253
254    Args:
255        params: Model parameters.
256        occ: Occupancy as a function of time in hours.
257
258    Returns:
259        ``f(t, C) = g occ(t) - lambda (C - C_out) + s``, ready for :func:`integrate`.
260
261    Example:
262        >>> f = co2_rhs(CO2Params(g=10.0, lam=1.0, c_out=400.0), lambda t: 2.0)
263        >>> f(0.0, 420.0)
264        0.0
265    """
266
267    def f(t, C):
268        """dC/dt [ppm/h] at time ``t`` [h] and concentration ``C`` [ppm]."""
269        return params.g * occ(t) - params.lam * (C - params.c_out) + params.s
270
271    return f

274def simulate_co2(params: CO2Params, t_h: np.ndarray, occ_values: np.ndarray, c0: float, method: str = "rk4") -> np.ndarray:
275    """Simulate CO2 given occupancy sampled on the same grid.
276
277    Occupancy is linearly interpolated between samples, so RK stages that fall
278    between grid points see a consistent input.
279
280    Args:
281        params: Model parameters.
282        t_h: Time grid in hours.
283        occ_values: Occupancy at ``t_h`` (persons; NaN treated as 0).
284        c0: Initial concentration [ppm].
285        method: Integrator (see :func:`integrate`), or ``"exact_linear"`` for
286            the closed-form step that is exact for piecewise-linear occupancy
287            (used for parameter fitting and as the verification reference).
288
289    Returns:
290        Simulated concentration on ``t_h``.
291
292    Raises:
293        ValueError: If ``method`` is unknown.
294
295    Example:
296        >>> p = CO2Params(g=50.0, lam=2.0, c_out=420.0)
297        >>> c = simulate_co2(p, np.linspace(0, 6, 61), np.full(61, 4.0), c0=420.0)
298        >>> round(float(c[-1]))  # steady state 420 + g N / lambda = 520
299        520
300    """
301    occ_values = np.nan_to_num(np.asarray(occ_values, dtype=float))
302    if method == "exact_linear":
303        return _simulate_linear_exact(params, t_h, occ_values, c0)
304
305    def occ(t):
306        """Occupancy at any time ``t`` by linear interpolation."""
307        return np.interp(t, t_h, occ_values)
308
309    return integrate(co2_rhs(params, occ), c0, t_h, method)[:, 0]

312def _simulate_linear_exact(p: CO2Params, t_h, occ, c0):
313    """Exact discretisation for piecewise-linear occupancy (verification reference).
314
315    For ``C' = -lam C + u(t)`` with ``u`` linear on ``[t_n, t_n+h]`` the step is
316    ``C_{n+1} = e^{-lam h} C_n + int_0^h e^{-lam (h-s)} u(t_n + s) ds`` (closed form).
317    """
318    C = np.empty(len(t_h))
319    C[0] = c0
320    for n in range(len(t_h) - 1):
321        h = t_h[n + 1] - t_h[n]
322        u0 = p.g * occ[n] + p.lam * p.c_out + p.s
323        u1 = p.g * occ[n + 1] + p.lam * p.c_out + p.s
324        a = np.exp(-p.lam * h)
325        # int_0^h e^{-lam(h-s)} ds = (1-a)/lam ; int_0^h e^{-lam(h-s)} s ds = h/lam - (1-a)/lam^2
326        I0 = (1 - a) / p.lam
327        I1 = h / p.lam - (1 - a) / p.lam**2
328        C[n + 1] = a * C[n] + u0 * I0 + (u1 - u0) / h * I1
329    return C

332def fit_equation_error(C: np.ndarray, occ: np.ndarray, h: float) -> CO2Params:
333    """Linear least-squares estimate of ``(g, lambda, C_out)`` from finite differences.
334
335    Rewrites the ODE as ``dC/dt = a N + b C + c`` (``a=g``, ``b=-lambda``,
336    ``c=lambda C_out``), estimates ``dC/dt`` by central differences and solves
337    the overdetermined linear system with ``numpy.linalg.lstsq`` (QR/SVD - more
338    stable than forming the normal equations).
339
340    Args:
341        C: CO2 series on a uniform grid (NaN rows are dropped).
342        occ: Occupancy on the same grid.
343        h: Grid spacing in hours.
344
345    Returns:
346        :class:`CO2Params` (with ``s = 0``). ``lambda`` is floored at 1e-3 /h
347        so the fitted ventilation is always positive.
348
349    Note:
350        Differencing noisy data puts noise into the regressor ``dC/dt`` and
351        biases ``lambda``; this estimate is therefore used only as the
352        starting guess for :func:`fit_output_error`.
353
354    Example:
355        >>> p = CO2Params(g=50.0, lam=2.0, c_out=420.0)
356        >>> t = np.arange(0, 48, 0.1)
357        >>> occ = np.where((t % 24 > 8) & (t % 24 < 17), 5.0, 0.0)
358        >>> C = simulate_co2(p, t, occ, 420.0, method="exact_linear")
359        >>> est = fit_equation_error(C, occ, h=0.1)
360        >>> bool(abs(est.c_out - 420) < 20)
361        True
362    """
363    dC = finite_difference(C, h, "central")
364    X = np.column_stack([occ, C, np.ones_like(C)])
365    ok = np.isfinite(dC) & np.isfinite(X).all(axis=1)
366    (a, b, c), *_ = np.linalg.lstsq(X[ok], dC[ok], rcond=None)
367    lam = max(-b, 1e-3)  # physical constraint: positive ventilation
368    return CO2Params(g=float(a), lam=float(lam), c_out=float(c / lam))

371def fit_output_error(
372    segments: list[tuple[np.ndarray, np.ndarray, np.ndarray]],
373    init: CO2Params,
374    method: str = "exact_linear",
375    fit_s: bool = False,
376) -> tuple[CO2Params, object]:
377    """Nonlinear least-squares fit by simulating each observed segment.
378
379    Minimises the sum of squared simulation errors over all segments, each
380    started from its own first observation. Bounds keep parameters physical
381    (``g >= 0``, ``0.01 <= lambda <= 20 /h``, ``300 <= C_out <= 600 ppm``).
382
383    Args:
384        segments: List of ``(t_h, C, occ)`` arrays without NaNs in ``C``.
385        init: Starting guess (e.g. from :func:`fit_equation_error`).
386        method: Integrator used inside the objective.
387        fit_s: Also fit the constant source ``s``.
388
389    Returns:
390        ``(params, scipy_result)`` where ``scipy_result`` is the
391        :class:`scipy.optimize.OptimizeResult` from ``least_squares``.
392
393    Example:
394        >>> true = CO2Params(g=50.0, lam=2.0, c_out=420.0)
395        >>> t = np.arange(0, 24, 1 / 6)
396        >>> occ = np.where((t > 8) & (t < 17), 5.0, 0.0)
397        >>> C = simulate_co2(true, t, occ, 420.0, method="exact_linear")
398        >>> fit, _ = fit_output_error([(t, C, occ)], CO2Params(30.0, 1.0, 400.0))
399        >>> round(fit.g, 3), round(fit.lam, 3), round(fit.c_out, 3)
400        (50.0, 2.0, 420.0)
401    """
402
403    def unpack(p):
404        """Turn the optimiser's parameter vector into :class:`CO2Params`."""
405        return CO2Params(float(p[0]), float(p[1]), float(p[2]), float(p[3]) if fit_s else 0.0)
406
407    def resid(p):
408        """Simulation-minus-measurement residuals over all segments."""
409        prm = unpack(p)
410        return np.concatenate([simulate_co2(prm, t, o, C[0], method) - C for t, C, o in segments])
411
412    x0 = [max(init.g, 0.0), np.clip(init.lam, 0.02, 19.0), np.clip(init.c_out, 301, 599)] + ([0.0] if fit_s else [])
413    lb = [0.0, 0.01, 300.0] + ([-500.0] if fit_s else [])
414    ub = [np.inf, 20.0, 600.0] + ([500.0] if fit_s else [])
415    res = least_squares(resid, x0, bounds=(lb, ub), x_scale="jac")
416    return unpack(res.x), res

419# --------------------------------------------------- gap filling: IVP & BVP
420def secant(fun: Callable[[float], float], x0: float, x1: float, tol: float = 1e-8, max_iter: int = 50):
421    """Secant root finder (derivative-free Newton).
422
423    Args:
424        fun: Scalar function whose root is wanted.
425        x0: First starting point.
426        x1: Second starting point.
427        tol: Stop when ``|fun(x)| < tol`` or the step is below ``tol * (1 + |x|)``.
428        max_iter: Iteration cap.
429
430    Returns:
431        ``(root, iterations)``.
432
433    Raises:
434        RuntimeError: If it fails to converge or the secant slope vanishes.
435
436    Example:
437        >>> root, k = secant(lambda x: x**2 - 2, 1.0, 2.0)
438        >>> round(root, 8)
439        1.41421356
440    """
441    f0, f1 = fun(x0), fun(x1)
442    for k in range(1, max_iter + 1):
443        if f1 == f0:
444            raise RuntimeError("secant slope is zero")
445        x2 = x1 - f1 * (x1 - x0) / (f1 - f0)
446        x0, f0, x1, f1 = x1, f1, x2, fun(x2)
447        if abs(f1) < tol or abs(x1 - x0) < tol * (1 + abs(x1)):
448            return x1, k
449    raise RuntimeError("secant method did not converge")

452def fill_ode(
453    y: np.ndarray,
454    occ: np.ndarray,
455    params: CO2Params,
456    h: float,
457    mode: str = "shooting",
458    method: str = "rk4",
459    return_iterations: bool = False,
460):
461    """Fill CO2 gaps by integrating the mass balance across each gap.
462
463    Args:
464        y: CO2 series with NaNs on a uniform grid.
465        occ: Occupancy on the same grid.
466        params: Fitted :class:`CO2Params`.
467        h: Grid spacing in hours.
468        mode: ``"ivp"`` integrates forward from the last observation before the
469            gap (uses only the left boundary). ``"shooting"`` solves the
470            two-point boundary-value problem: it finds the constant source ``s``
471            for which the trajectory starting at the left value lands on the
472            first observation after the gap (secant iteration), so the fill is
473            continuous at both ends.
474        method: Integrator.
475        return_iterations: Also return the list of secant iteration counts.
476
477    Returns:
478        Filled copy of ``y`` (and iteration counts). A gap at the end of the
479        series has no right boundary, so it is always integrated as an IVP; a
480        gap at the start has no left boundary, so it holds the first
481        observation.
482
483    Raises:
484        ValueError: If ``mode`` is not ``"ivp"`` or ``"shooting"``.
485        RuntimeError: If the shooting iteration fails to converge.
486
487    Example:
488        >>> p = CO2Params(g=50.0, lam=2.0, c_out=420.0)
489        >>> t = np.arange(0, 12, 1 / 6)
490        >>> occ = np.where((t > 3) & (t < 9), 5.0, 0.0)
491        >>> truth = simulate_co2(p, t, occ, 420.0, method="exact_linear")
492        >>> y = truth.copy()
493        >>> y[20:40] = np.nan
494        >>> filled = fill_ode(y, occ, p, h=1 / 6)
495        >>> bool(np.max(np.abs(filled - truth)) < 0.5)
496        True
497    """
498    from .interpolation import fill_locf, gap_spans
499
500    if mode not in ("ivp", "shooting"):
501        raise ValueError("mode must be 'ivp' or 'shooting'")
502    y = np.array(y, dtype=float, copy=True)
503    miss = np.isnan(y)
504    iters = []
505    for a, b in gap_spans(miss):
506        if a == 0:
507            continue
508        t = np.arange(a - 1, min(b + 1, y.size)) * h
509        o = occ[a - 1 : min(b + 1, y.size)]

def integrate( f: Callable[[float, numpy.ndarray], numpy.ndarray], y0, t: numpy.ndarray, method: str = 'rk4', newton_tol: float = 1e-10, newton_max: int = 50) -> numpy.ndarray: 
104def integrate(
105    f: Callable[[float, np.ndarray], np.ndarray],
106    y0,
107    t: np.ndarray,
108    method: str = "rk4",
109    newton_tol: float = 1e-10,
110    newton_max: int = 50,
111) -> np.ndarray:
112    """Integrate ``y' = f(t, y)`` on the given time grid with a fixed-step method.
113
114    Args:
115        f: Right-hand side; takes a float time and a 1-D state, returns an array.
116        y0: Initial state (scalar or 1-D).
117        t: Increasing time grid; the step may vary.
118        method: ``"euler"`` (explicit, order 1), ``"heun"`` (explicit RK2),
119            ``"midpoint"`` (explicit RK2), ``"rk4"`` (classical RK, order 4),
120            ``"implicit_euler"`` (backward Euler, order 1, A-stable) or
121            ``"trapezoid"`` (implicit, order 2, A-stable).
122        newton_tol: Relative step-size tolerance of the Newton iteration used
123            by the implicit methods (Jacobian by forward finite differences).
124        newton_max: Maximum Newton iterations per implicit step.
125
126    Returns:
127        Array of shape ``(len(t), dim)``.
128
129    Raises:
130        RuntimeError: If the implicit Newton iteration fails to converge.
131        ValueError: If ``method`` is unknown.
132
133    Example:
134        >>> t = np.linspace(0.0, 1.0, 11)
135        >>> Y = integrate(lambda t, y: -y, 1.0, t, method="rk4")  # exact: e^-1
136        >>> round(float(Y[-1, 0]), 6)
137        0.36788
138    """
139    y0 = np.atleast_1d(np.asarray(y0, dtype=float))
140    Y = np.empty((len(t), y0.size))
141    Y[0] = y0
142    for n in range(len(t) - 1):
143        h, tn, yn = t[n + 1] - t[n], t[n], Y[n]
144        if method == "euler":
145            Y[n + 1] = yn + h * f(tn, yn)
146        elif method == "heun":
147            k1 = f(tn, yn)
148            k2 = f(tn + h, yn + h * k1)
149            Y[n + 1] = yn + h / 2 * (k1 + k2)
150        elif method == "midpoint":
151            k1 = f(tn, yn)
152            Y[n + 1] = yn + h * f(tn + h / 2, yn + h / 2 * k1)
153        elif method == "rk4":
154            k1 = f(tn, yn)
155            k2 = f(tn + h / 2, yn + h / 2 * k1)
156            k3 = f(tn + h / 2, yn + h / 2 * k2)
157            k4 = f(tn + h, yn + h * k3)
158            Y[n + 1] = yn + h / 6 * (k1 + 2 * k2 + 2 * k3 + k4)
159        elif method in ("implicit_euler", "trapezoid"):
160            theta = 1.0 if method == "implicit_euler" else 0.5
161            fn = f(tn, yn)
162
163            def G(z, yn=yn, h=h, tn=tn, fn=fn, theta=theta):
164                """Residual of the implicit (theta-method) update; zero at the new state."""
165                return z - yn - h * (theta * f(tn + h, z) + (1 - theta) * fn)
166
167            z = yn + h * fn  # explicit predictor
168            for _ in range(newton_max):
169                g = G(z)
170                eps = 1e-7 * (1 + np.abs(z))
171                J = np.column_stack([(G(z + e) - g) / e[i] for i, e in enumerate(np.diag(eps))])
172                dz = np.linalg.solve(J, -g)
173                z = z + dz
174                if np.max(np.abs(dz)) < newton_tol * (1 + np.max(np.abs(z))):
175                    break
176            else:
177                raise RuntimeError("Newton iteration did not converge in implicit step")
178            Y[n + 1] = z
179        else:
180            raise ValueError(f"unknown method {method!r}")
181    return Y

def secant( fun: Callable[[float], float], x0: float, x1: float, tol: float = 1e-08, max_iter: int = 50): 
421def secant(fun: Callable[[float], float], x0: float, x1: float, tol: float = 1e-8, max_iter: int = 50):
422    """Secant root finder (derivative-free Newton).
423
424    Args:
425        fun: Scalar function whose root is wanted.
426        x0: First starting point.
427        x1: Second starting point.
428        tol: Stop when ``|fun(x)| < tol`` or the step is below ``tol * (1 + |x|)``.
429        max_iter: Iteration cap.
430
431    Returns:
432        ``(root, iterations)``.
433
434    Raises:
435        RuntimeError: If it fails to converge or the secant slope vanishes.
436
437    Example:
438        >>> root, k = secant(lambda x: x**2 - 2, 1.0, 2.0)
439        >>> round(root, 8)
440        1.41421356
441    """
442    f0, f1 = fun(x0), fun(x1)
443    for k in range(1, max_iter + 1):
444        if f1 == f0:
445            raise RuntimeError("secant slope is zero")
446        x2 = x1 - f1 * (x1 - x0) / (f1 - f0)
447        x0, f0, x1, f1 = x1, f1, x2, fun(x2)
448        if abs(f1) < tol or abs(x1 - x0) < tol * (1 + abs(x1)):
449            return x1, k
450    raise RuntimeError("secant method did not converge")
